StudySession

@Model final class StudySession {
    var id: UUID = UUID()
    var startTime: Date = Date()
    var endTime: Date?
    var duration: TimeInterval = 0
    var subject: Subject?

    var date: Date { Calendar.current.startOfDay(for: startTime) }

    // 抜粋: 「詳しく記録する」の項目
    var note: String?            // メモ。v1.4 から計測を終えたあとの画面のメモもここに入り、週のふりかえりの「この週のメモ」に出る
    var nextStudyDate: Date?     // 旧「次回やること」。v1.4 からは書かない・表示に使わない（値は残す）
    var nextStudyNote: String?   // 同上
}
v1.4: 旧「次回やること」（nextStudyDate / nextStudyNote）は「やること」に一本化した。 起動時に1回だけ、直近7日以降の予定をやること（origin = migrated）へ写す（NextStudyMigrationService。元の値は消さない）。 手動記録の「次回の予定」は v1.4 からやることとして作り、この2項目には書かない。

StudyTask

@Model final class StudyTask {
    var id: UUID = UUID()
    var title: String = ""
    var isCompleted: Bool = false
    var dueDate: Date? = nil             // 期限日（v1.2〜。nil の場合は旧タスクまたは期限なし）
    var subject: Subject? = nil          // 紐づく科目（v1.3〜。学習計画生成時は必須。nil = 科目未設定）
    var linkedSessionID: UUID? = nil     // 「記録して完了」時に紐づく StudySession.id
    var completedAt: Date? = nil         // 完了日時
    var sortOrder: Int = 0
    var createdAt: Date = Date()

    // 後方互換フィールド（v1.2 以降は非推奨）
    var periodRawValue: String = "day"   // 旧: TaskPeriod.rawValue: "day" / "week" / "month"
    var periodStart: Date = Date()       // 旧: 期間開始（例: 今日0時）
    var periodEnd: Date = Date()         // 旧: 期間終了（例: 今日23:59:59）

    // v1.4 で追加（すべて既定値つき。既存データはこの値で読み込まれる）
    var planID: UUID?                                          // 学習計画から作られたやることの計画ID（計画以外は nil）
    var planAmount: Int?                                       // 学習計画で割り当てた量（進み具合の計算用）
    var dueScopeRawValue: String = TaskDueScope.day.rawValue   // "day" / "week"（week のとき dueDate はその週の日曜）
    var originRawValue: String = TaskOrigin.manual.rawValue    // manual / plan / next / carried / migrated
    var handoffNote: String?                                   // 次に開いたとき「前回のメモ」として見せる
}

enum TaskDueScope: String { case day, week }                            // v1.4
enum TaskOrigin: String { case manual, plan, next, carried, migrated }  // v1.4

// TaskPeriod は後方互換のため残す（v1.2 以降は非推奨）
enum TaskPeriod: String {
    case day   = "day"
    case week  = "week"
    case month = "month"
}

StudyPlan（v1.4 新規）

/// 学習計画。「教材の総量を、終了日までの日数に配る」計画そのもの。
/// 日ごとのやること（StudyTask）は planID でこの計画に紐づく。
/// v1.3 以前に作った計画はこのモデルを持たない（やることだけが残る）ため、進み具合には出ない。
@Model final class StudyPlan {

    var subject: Subject?
    var totalAmount: Int = 0
    var unit: String = ""
    var startDate: Date = Date()
    var endDate: Date = Date()
    var splitModeRawValue: String = PlanSplitMode.daily.rawValue   // daily / weekdayFilter / single
    var weekdaysText: String = ""        // 曜日指定のときに配る曜日（1=日〜7=土 をカンマ区切り）
    var createdAt: Date = Date()
    var isArchived: Bool = false         // 一覧から外したか（終わった計画・やめた計画）
}

計算と書き込みは 学習計画の進み具合と配り直し を参照。modelContainer に StudyPlan.self を追加した。

v1.4 のデータ変更のまとめ

すべて追加のみ。既存項目の削除・型変更・名前変更はしていない。 StudyTask に5項目（既定値つき）を足し、StudyPlan を足しただけなので、SwiftData の自動の移行で開ける。 1.3.4 のデータの形を写したテストで、科目・学習記録・やること・成績・教材・目標・設定・週次レポートの件数と中身が変わらないこと、 足した項目が既定値で読めること、旧「次回やること」が元の値を残したままやることへ写ることを確かめた。

科目を消したときの参照（SS-022）

Subject は StudySession（cascade）と TestRecord（nullify）には削除の決まりを持つが、StudyTask・StudyPlan・StudyMaterial からの参照には持たない。 そのまま科目を消すと消えた科目を指したまま残り、科目名を読んだ瞬間に落ちる（以前からの問題。v1.4 のホームは起動直後に科目名を読むので落ちうる）。 v1.4 では SubjectReferenceService で、科目を消す前に3つの参照を外し、起動時（RootView で本編を出す前）に消えた科目への参照を外す。削除の決まり自体は変えていない。

TestRecord（v1.1.0 新規）

@Model final class TestRecord {
    var id: UUID = UUID()
    var date: Date = Date()
    var testTypeRaw: String = TestType.regular.rawValue  // TestType.rawValue を保存
    var subject: Subject? = nil                           // nullify リレーション
    var score: Double = 0.0
    var maxScore: Double = 100.0
    var deviation: Double? = nil   // 模試のみ使用
    var rank: Int? = nil           // 任意
    var memo: String? = nil        // 任意
    var createdAt: Date = Date()

    // Computed properties
    var scoreRate: Double { maxScore > 0 ? score / maxScore * 100 : 0 }
    var testType: TestType { TestType(rawValue: testTypeRaw) ?? .other }
}

enum TestType: String, Codable, CaseIterable {
    case regular = "regular"  // 定期テスト
    case mock    = "mock"     // 模試
    case quiz    = "quiz"     // 小テスト
    case other   = "other"    // その他

    var displayName: String {
        switch self {
        case .regular: return "定期テスト"
        case .mock:    return "模試"
        case .quiz:    return "小テスト"
        case .other:   return "その他"
        }
    }
}

WeeklyCoachReport（v1.2.0 新規）

/// 週次学習レポート（Premium機能「先週のふりかえり」）。
/// アプリ起動時に前週分を集計して1レコード保存する（記録ゼロ週は生成しない）。
/// コーチ文言は保存せず View 側で導出する。
@Model final class WeeklyCoachReport {
    var id: UUID = UUID()

    /// 対象週の月曜の学習日開始インスタント（重複生成チェックのキー）
    var weekStart: Date = Date()

    /// 週の合計学習時間（秒）
    var totalSeconds: Double = 0

    /// 前々週の合計学習時間（秒）。前週比の算出用
    var previousWeekSeconds: Double = 0

    /// 最も学習時間が多かった科目名（記録なしなら空文字）
    var topSubjectName: String = ""

    /// 最多科目が全体に占める割合（0〜100）
    var topSubjectPercent: Double = 0

    /// 週のうち勉強した日数（0〜7）
    var studyDayCount: Int = 0

    /// 生成時点での試験日までの残日数（試験日未設定なら nil）
    var examDaysRemaining: Int?

    /// 生成時点での1日の目標分数（未設定なら nil。達成率の算出用）
    var dailyGoalMinutes: Int?

    var generatedAt: Date = Date()

    // Computed properties
    /// 前週比の増減率（%）。前々週が0なら nil（比較不能）
    var changePercent: Double? {
        guard previousWeekSeconds > 0 else { return nil }
        return ((totalSeconds - previousWeekSeconds) / previousWeekSeconds) * 100
    }

    /// 1日目標に対する達成率（0〜100超もあり得る）。目標未設定なら nil
    var goalAchievementPercent: Double? {
        guard let goal = dailyGoalMinutes, goal > 0 else { return nil }
        let weeklyGoalSeconds = Double(goal) * 7 * 60
        return totalSeconds / weeklyGoalSeconds * 100
    }
}

マイグレーション方針

SwiftDataの@Modelは全プロパティにデフォルト値必須。新規プロパティ追加時は必ずデフォルト値を設定する。
